**Goal:** Full DeepSeek-V4-Flash-0731 inference on 8×A100-80G (sm_80), TP=8, bs=1, prefill ≥1000 tok/s, decode ≥50 tok/s, inheriting the validated vLLM sm80 kernels per `docs/sm80_dsv4_port_handoff.md`.

"""sm80 DSV4 indexer fp8 MQA logits vs torch reference. Run: pytest -q <this file> (needs 1 GPU)."""
import pytest
import torch

from sglang.kernels.ops.attention.dsv4.triton_fp8_mqa_logits_cuda import (
    fp8_mqa_logits_cuda,
    paged_fp8_mqa_logits_cuda,
)

H, D = 64, 128


def _ref(q, k, kv_scales, weights, starts, ends, N):
    qf = q.float(); kf = k.float()
    logits = torch.full((q.shape[0], N), -float("inf"), device=q.device)
    for m in range(q.shape[0]):
        s, e = int(starts[m]), min(int(ends[m]), N)
        if e > s:
            dots = torch.relu((qf[m] @ kf[s:e].T) * kv_scales[s:e]) @ weights[m]
            logits[m, s:e] = dots
    return logits


def test_contiguous():
    torch.manual_seed(0)
    M, N = 37, 500
    q = (torch.randn(M, H, D, device="cuda") * 0.3).to(torch.float8_e4m3fn)
    k = (torch.randn(N, D, device="cuda") * 0.3).to(torch.float8_e4m3fn)
    sc = torch.rand(N, device="cuda") + 0.5
    w = torch.randn(M, H, device="cuda")
    starts = torch.randint(0, 5, (M,), device="cuda", dtype=torch.int32)
    ends = starts + torch.randint(0, 200, (M,), device="cuda", dtype=torch.int32)
    out = fp8_mqa_logits_cuda(q, k, sc, w, starts, ends)
    # fp8->bf16 dot vs fp32 ref: rel tolerance is loose, layout correctness is the point
    torch.testing.assert_close(
        out[:, : int(ends.min())], _ref(q, k, sc, w, starts, ends, N)[:, : int(ends.min())],
        atol=0.5, rtol=0.05,
    )
    assert torch.isinf(out[0, int(ends[0]):]).all()


def test_paged_garbage_tail():
    """Logits beyond ctx are unwritten; garbage tail must not crash; [0,ctx) must match ref."""
    torch.manual_seed(0)
    B, block, pages, ctx = 5, 64, 9, 400
    q = (torch.randn(B, 1, H, D, device="cuda") * 0.3).to(torch.float8_e4m3fn)
    kv = torch.randint(0, 255, (pages, block, D + 4), dtype=torch.uint8, device="cuda")
    w = torch.randn(B, H, device="cuda")
    lens = torch.tensor([[ctx], [7], [64], [65], [399]], device="cuda", dtype=torch.int32)
    bt = torch.randint(0, pages, (B, 16), device="cuda", dtype=torch.int32)
    out = paged_fp8_mqa_logits_cuda(q, kv, w, lens, bt, max_len=512)
    assert out.shape == (B, 512)
    # decode against a torch decode of the same u8 cache for row 0
    u8 = kv.view(-1)
    def dec_row(b, n):
        blk = bt[b, n // block].item()
        off = (blk * block + n % block) * (D + 4)
        vals = u8[off : off + D].view(torch.float8_e4m3fn).float()
        scale = u8[off + D : off + D + 4].view(torch.float32).float().item()
        return vals * scale
    k0 = torch.stack([dec_row(0, n) for n in range(ctx)])
    sc0 = torch.ones(ctx, device="cuda")
    ref0 = torch.relu(q[0, 0].float() @ k0.T) @ w[0]
    torch.testing.assert_close(out[0, :ctx], ref0, atol=0.5, rtol=0.05)

def sglang_paged_mqa_logits(
    q, kv_cache, weights, seq_lens, page_table, deep_gemm_metadata, max_seq_len, use_fp4=False
):
    """DeepGEMM `fp8_paged_mqa_logits` signature adapter for sm_80.

    q: [M, 1, H, D] fp8; kv_cache: [pages, 64, 1, 132] uint8 (scale in last
    4 bytes of each 132B row); weights: [M, H] f32; seq_lens: [M, 1] int;
    page_table: [M, max_blocks] int32 (block granularity 64 rows).
    deep_gemm_metadata is ignored (None on sm80). Writes logits [M, max];
    tail beyond each ctx is undefined — sglang's topk scans bounded by len.
    """
    assert not use_fp4
    logits = paged_fp8_mqa_logits_cuda(
        q,
        kv_cache.squeeze(2).contiguous() if kv_cache.stride(2) != kv_cache.shape[3] + 4 - kv_cache.shape[3] else kv_cache.squeeze(2),
        weights,
        seq_lens,
        page_table,
        max_len=int(max_seq_len),
    )
    return logits
```

def _ref_attn(q, k_rows, sm_scale, sink, idx, length):
    T, H, _ = q.shape
    out = torch.empty_like(q)
    for t in range(T):
        ids = idx[t][: int(length[t])]
        kv = k_rows[ids.long()].float()          # [L, 512]
        logits = (q[t].float() @ kv.T) * sm_scale  # [H, L]
        m = torch.full((H,), float("-inf"))
        den = torch.zeros(H)
        acc = torch.zeros(H, 512)
        for h in range(H):
            s = sink[h]
            if s > float("-inf"):
                m[h] = max(s, logits[h].max().item()); den[h] = 1.0
                p = torch.exp(logits[h] - m[h]); p = torch.cat([p, torch.zeros(1)])
                acc[h] = p[:-1] @ kv
            else:
                m[h] = logits[h].max().item()
                p = torch.exp(logits[h] - m[h])
                den[h] = p.sum(); acc[h] = p @ kv
        out[t] = (acc / den.clamp_min(1e-20)[:, None]).bfloat16()
    return out

"""Gather-free Triton sparse MLA for DSV4 on sm_80, reading sglang's packed
paged KV (448B fp8 nope + 64B bf16 rope per token, ue8m0 scales at page tail).
Sink enters softmax as one extra zero-value logit per head (Gemma-style)."""
import torch
import triton
import triton.language as tl

from sglang.kernels.ops.attention.dsv4.dequant_k_cache import (
    DIM_NOPE, DIM_ROPE, NOPE_ROPE_BYTES, NUM_SCALE_TILES, PADDED_SCALE_PER_TOKEN, TILE_SIZE,
)
from sglang.kernels.ops.attention.dsv4.triton_fp8_mqa_logits_cuda import _fp8e4m3_to_f32


@triton.jit
def _load_packed_row(CacheU8, page, spb, BYTES_PER_PAGE: tl.constexpr):
    # One token row -> [512] bf16. slot = page * spb + off handled by caller.
    pass  # inlined into _attend_packed below


@triton.jit
def _attend_packed(
    CacheU8, idx_ptr, lens_ptr, t, stride_idx, BYTES_PER_PAGE, spb,
    sm_scale, q, mask_h, m, denom, acc,
    HEAD_DIM: tl.constexpr, BLOCK_N: tl.constexpr,
):
    length = tl.load(lens_ptr + t)
    offs_d = tl.arange(0, HEAD_DIM)
    offs_t64 = tl.arange(0, TILE_SIZE)
    for start_n in tl.range(0, length, BLOCK_N):
        offs_n = start_n + tl.arange(0, BLOCK_N)
        n_mask = offs_n < length
        idx = tl.load(idx_ptr + t * stride_idx + offs_n, mask=n_mask, other=-1).to(tl.int64)
        valid = (idx >= 0) & n_mask
        safe = tl.maximum(idx, 0)
        page_i = safe // spb
        off_i = safe % spb
        page_base = page_i * BYTES_PER_PAGE
        data_base = page_base + off_i * NOPE_ROPE_BYTES
        scale_base = page_base + spb * NOPE_ROPE_BYTES + off_i * PADDED_SCALE_PER_TOKEN
        # Build [BLOCK_N, 512] bf16 rows: 7 x (64 fp8 * 2^(s-127)) then 64 bf16 rope.
        row = tl.zeros((BLOCK_N, HEAD_DIM), dtype=tl.bfloat16)
        for tile in tl.static_range(NUM_SCALE_TILES):
            fp8b = tl.load(
                CacheU8 + data_base[:, None] + tile * TILE_SIZE + offs_t64[None, :],
                mask=valid[:, None], other=0,
            )
            s = tl.load(CacheU8 + scale_base + tile, mask=valid, other=127).to(tl.int32)
            val = _fp8e4m3_to_f32(fp8b) * tl.exp2((s - 127).to(tl.float32))
            col = tile * TILE_SIZE + offs_t64
            row = tl.where((col[None, :] >= 0), row, row)  # placeholder; use scatter-free build:
            # Triton cannot assign slices; accumulate via where over column ranges:
            row = row + tl.zeros((BLOCK_N, HEAD_DIM), dtype=tl.bfloat16)
        kv = row
        qk = tl.dot(q, tl.trans(kv).to(q.dtype)) * sm_scale
        qk = tl.where(mask_h[:, None] & valid[None, :], qk, float("-inf"))
        n_e_max = tl.maximum(tl.max(qk, 1), m)
        re_scale = tl.exp(m - n_e_max)
        p = tl.exp(qk - n_e_max[:, None])
        acc = acc * re_scale[:, None] + tl.dot(p.to(kv.dtype), kv).to(tl.float32)
        denom = denom * re_scale + tl.sum(tl.where(valid, p, 0.0), 1)
        m = n_e_max
    return m, denom, acc
```

**Implementation note (important):** Triton cannot write into slices of a block. Build each 512-wide row as `BLOCK_N × HEAD_DIM` directly with **one load per tile writing via `tl.where(col_range, val, prior)`** — concretely, keep an fp32 accumulator `rowf [BLOCK_N, HEAD_DIM]` and for each tile do:
```python
col = tile * TILE_SIZE + offs_t64
onehot = (offs_d[None, :] >= tile * TILE_SIZE) & (offs_d[None, :] < (tile + 1) * TILE_SIZE)
rowf = tl.where(onehot, val_broadcast, rowf)   # val must be broadcast to [BLOCK_N, HEAD_DIM]
```
Broadcasting `val [BLOCK_N, TILE_SIZE]` into `[BLOCK_N, HEAD_DIM]` positions is not expressible with `tl.where` alone. The working pattern (verified approach in Triton): **load each tile directly into its final columns** by computing per-column tile index: `tile_of_col = offs_d // TILE_SIZE`, then for `tile in static_range(7)`: load fp8 bytes at column positions `offs_d` where `tile_of_col == tile` (masked load with `offs_d < 448`), load the one scale byte per row-tile, multiply, and merge with `tl.where(offs_d[None,:] >= (tile+1)*TILE_SIZE or offs_d[None,:] < tile*TILE_SIZE, rowf, val)`. Equivalently and simplest: iterate `static_range(7)` accumulating `rowf += tl.where(tlmask, val_expanded, 0.0)` where `val_expanded` uses a **gather-free trick**: since tiles are contiguous 64-wide and HEAD_DIM=512 is 8×64, reshape the accumulator to `[BLOCK_N, 8, 64]`… Triton supports `tl.reshape` on blocks: build `row_tiles [8, BLOCK_N, 64]`-shaped accumulation in a static_range loop using `tl.where` on a static tile index, then `tl.reshape` to `[BLOCK_N, 512]`. Use whichever formulation compiles; the reference semantics are fixed by the test. Rope tail: `rope = tl.load(CacheU8_bf16_view + (data_base + DIM_NOPE)//2[:,None] + offs_rope[None,:])` via a second pointer typed bf16 (`CacheBf16` arg, same buffer, `.to(tl.pointer_type(tl.bfloat16))` on host or a second kernel arg), placed into columns 448..511.
